    end

grads = {}

grad = {yb=0,yc=0,t=0,d=0,hb=0,hc=0}

function grad:new(_y,_h,_c)
    local obj = {y=_y, h=_h, col=_c}
    setmetatable(obj,{__index = self})
    add(grads,obj)
end

function grad:update()
    if self.t != self.d then
        self.t += 1
        local t = self.t / self.d
        self.y = -self.yc*t*(t-2)+self.yb
        self.h = -self.hc*t*(t-2)+self.hb
    end
end

function grad:draw(start)
    local y = self.y & 0xfffe.0000
 local    quarter = max(self.h>>2, 2)
 local start = start or 0
 local c1 = color()
 local col = self.col | (c1<<4)
 fillp()
 rectfill(4,start,123,y-1,c1)
    fillp(0x5f5f)
    rectfill(4,y,123,y+quarter-1,col)
    fillp(0x5a5a)
    rectfill(4,y+quarter,123,y+3*quarter-1,col)
    fillp(0x0a0a)
    rectfill(4,y+quarter*3,123,y+self.h-1,col)
    start = y+self.h
    return start
end

function grad:move_to(_yto,_hto,_d)
    local b = self.y 
    self.yb = b
    self.yc = _yto - b

    b = self.h
    self.hb = b
    self.hc = _hto - b

    self.t = 0
    self.d = _d
end

prop = {t = 0,b = 0,c = 0}

function prop:new(sx,sy,sw,sh,dx,dy,p,to,d,s)
    local obj = {sx = sx,
    sy = sy,
    sw = sw,
    sh = sh,
    dx = dx,
    x = dx,
    dy = dy,
    p = p,
    d = d&0xfffe,
    s = s,
    to = to}
    setmetatable(obj,{__index = self})
    add(props,obj,1)
end

function prop:update()
    if self.t == 0 then
        local b = self[self.p]
        self.c = self.to-b
        self.b = b
    end
    if self.t != self.d then
        self.t += 1
        self[self.p] = self.c*self.t/self.d+self.b
        if (self.s == 1) self.dx = self.x + (self.t%10\5)
    end
end

function prop:draw()
    sspr(self.sx,self.sy,self.sw,self.sh,self.dx,self.dy)
end

function pyoro:update()
    if self.state == 0 then
        if self.eating then
            self.a = (self.a + 1) % 32766
            self.sprite = self.a % 8 \ 4 * 36 + 8
            if (self.a == 40) self.eating = false
        end

        if btnp(5) or btnp(4) then
            self.sprite = 32
            self.state = 1
            self.tongue.x = self.x+2+6*self.flip_x
            self.tongue.y = self.y-1
            self.is_walking = false
           
         else
            local speed = 0.8 + (gamespeed_timer>>14)
             local vx = (btn(1) and speed or 0) - (btn(0) and speed or 0)
            self:move(vx)
            self.x = mid(4,self.x,112)
            if btn(1) or btn(0) then
                self.a = (self.a + 1) % 32766
                self.flip_x = (btn(0) and -1 or 1)
                self.eating = false
                self.sprite = self.a % 6 \ 3 * 12 + 8
                psfx(6,0,3)
                self.is_walking = true
            elseif self.is_walking then
                psfx(6,3,3)
                self.is_walking = false
            end
        end
    elseif self.state == 1 then
        if btn(5) or btn(4) then
            local speed = 1.5 + (gamespeed_timer>>12)
            self.tongue.x += speed*self.flip_x
            self.tongue.y -= speed
        else
            self.state = 2
        end

        for o in all(objects) do
            self.eating = o.base == seed and overlaps(self.tongue,o)
            if self.eating then
                self.state = 2
                o:get_licked(self)
                self.licked_seed = o
                break
            end
        end

        if self.tongue.l == 3 then
            psfx(52,0,8)
        end

        if self.tongue.x != mid(5,self.tongue.x,116) or self.tongue.y <= 10 then
            self.state = 2
        end
    elseif self.state == 2 then
            local speed = 6 + (gamespeed_timer>>14)
            self.tongue.x -= speed*self.flip_x
            self.tongue.y += speed
            if self.tongue.l <= 1 then
                self.state = 0
                psfx(6,16,8)
                self.a = 0
                self.sprite = 8
                del(objects,self.licked_seed)
                if (self.licked_seed and self.licked_seed.on_eat) self.licked_seed.on_eat(self)
               
                self.licked_seed = nil
            end
    elseif self.state == 3 then
        self.y += 0.25
        if self.y > 118 then
            del(objects,self)
            game_state = 2
            music(26)
        end
    end

    if self.state != 3 then
        if floor[(self.x+2)\6+1] != 1 then
            self:die()
        else
            for o in all(objects) do
                if o.base == seed and overlaps(self,o) then
                    smoke:new(o.x,o.y)
                    del(objects,o)
                    self:die()
                    break
                end
            end
        end
    end

    self.tongue.l = (abs(self.x-self.tongue.x)-3.5-1.5*self.flip_x)\6
end

function pyoro:die()
    self.state = 3
    self.sprite = 56 
    psfx(52,8,8,48)
    music(-1)
end

    --print(self.tongue.l,self.x, self.y-6,0) -- Debug
    sspr(self.sprite,0,12,12,self.x,self.y,12,12,flip)

function pyoro:move(x)    --compare to the movement code article from May 2020
    self.remainder_x += x
    local mx = flr(self.remainder_x + 0.5)
    self.remainder_x -= mx

    local total = mx
    local mxs = sgn(mx)
    while mx != 0
    do
        if floor[(self.x+mxs+2)\6+1] != 1 then --values for floor are 0 (no floor), 1 (floor), and 2(angel about to replace floor)
            return true
        else
            self.x += mxs
            mx -= mxs
        end
    end

    return false
end

function seed:update()
    if self.state == 0 then
        if (timer - self.timer) == self.anim[self.a*2-1] then
            self.a = self.a%4+1    --sets animation frame
            self.timer = timer
        end
        self.c = timer % (#self.color*2) \ 2 + 1 -- sets color palatte for blue seed
        self.y += self.speed
        if (self.y >= 118) del(objects,self)
    elseif self.state == 1 then
        self.y = self.attatched.y-6
        self.x = (self.a == 3 and self.attatched.x+2 or self.attatched.x-7)
        if self.attatched.l == 0 then
            del(objects,self)
        end
    end
   
    if self.y > 102 then
        local x1 = (self.x - 1) \ 6 + 1
        local x2 = (self.x + 4) \ 6 + 1
        if (floor[x1] == 1) then
            floor[x1] = 0

            floor[x2] = 0
            smoke:new(self.x,self.y)

function overlaps(a, b, ox, oy) -- hitbox collision
    if a == b then return false end
    ox = ox or 0
    oy = oy or 0
    return
        ox + a.x + a.hit_x + a.hit_w > b.x + b.hit_x and
        oy + a.y + a.hit_y + a.hit_h > b.y + b.hit_y and
        ox + a.x + a.hit_x < b.x + b.hit_x + b.hit_w and
        oy + a.y + a.hit_y < b.y + b.hit_y + b.hit_h
end
